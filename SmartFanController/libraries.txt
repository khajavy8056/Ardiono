# ============================================================================
#  فهرست کتابخانه‌ها و وابستگی‌های پروژه «کنترل هوشمند دما و سرعت فن»
# ============================================================================
#  این فهرست برای نصب دستی در Arduino IDE یا با arduino-cli استفاده می‌شود:
#
#    arduino-cli lib install "DHT sensor library" "Adafruit Unified Sensor" \
#        "LiquidCrystal I2C" "Adafruit SSD1306" "Adafruit GFX Library" \
#        "Adafruit BusIO"
#
#  نسخه‌های زیر، نسخه‌هایی هستند که پروژه با آن‌ها کامپایل و تست شده است.
# ----------------------------------------------------------------------------

# --- سنسور دما DHT11 / DHT22 ---
DHT sensor library (Adafruit)        1.4.7     # خواندن DHT11 و DHT22
Adafruit Unified Sensor              1.1.15    # وابستگی لازم برای DHT sensor library

# --- نمایشگر LCD 16x2 با I2C ---
LiquidCrystal I2C (Frank de Brabander) 1.1.3   # کنترل LCD 16x2 از طریق I2C

# --- نمایشگر OLED 128x64 با I2C ---
Adafruit SSD1306                     2.5.17    # درایور نمایشگر OLED
Adafruit GFX Library                 1.13.0    # کتابخانه گرافیکی پایه
Adafruit BusIO                       1.17.4    # لایه I2C/SPI (وابستگی SSD1306)
